# PHP

# Telegram Bot (Python)
import requests, time
order = "ORD123456"
while True:
    r = requests.get(f"https://your-app.onrender.com/api/v2/check/YOUR_KEY?order_id={order}")
    d = r.json()
    if d["status"] == "PAID":
        print("✅ Paid!", d["amount"]); break
    time.sleep(5)
